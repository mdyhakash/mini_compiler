// Example program for Mini Compiler
// Demonstrates variables, arithmetic, and control flow

int x = 10;
int y = 5;
int result;

result = x * y;

print(result);

if (result > 20) {
    int temp = result - 10;
    print(temp);
}

int count = 0;

while (count < 4) {
    print(count);
    count = count + 1;
}

return result;
